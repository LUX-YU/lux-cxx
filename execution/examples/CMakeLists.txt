foreach(example main_loop cancellation two_contexts)
    add_executable(lux_execution_${example} ${example}.cpp)
    target_link_libraries(lux_execution_${example} PRIVATE lux::cxx::execution)
    lux_execution_test_options(lux_execution_${example})
    if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
        add_test(NAME execution.example.${example} COMMAND lux_execution_${example})
        set_tests_properties(execution.example.${example} PROPERTIES TIMEOUT 60)
    endif()
endforeach()
